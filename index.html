function ask(question) {
  const q = question.toLowerCase();
  let answer = "Попробуй уточнить вопрос. Пока я лучше всего отвечаю на вопросы про HTML, CSS, JavaScript, ошибки и запуск кода.";
  if (q.includes("html")) answer = lessons[0].answer;
  else if (q.includes("css") || q.includes("цвет") || q.includes("дизайн")) answer = lessons[2].answer;
  else if (q.includes("javascript") || q.includes(" js") || q.includes("функци")) answer = lessons[4].answer;
  else if (q.includes("переменн") || q.includes("числ")) answer = lessons[5].answer;
  else if (q.includes("ошиб") || q.includes("не работает")) answer = "Не паникуй! Проверь: 1) закрыты ли кавычки; 2) есть ли закрывающие теги; 3) нажал ли ты «Запустить». Если не помогло, пришли мне свой код — разберём вместе.";
  else if (q.includes("запустить") || q.includes("результат")) answer = "Напиши код в тёмном редакторе и нажми синюю кнопку «Запустить». Результат появится ниже, в белом окне.";
  else if (q.includes("начать") || q.includes("учить")) answer = "Начни с первого урока HTML. Измени текст в примере, нажми «Запустить» и посмотри, что поменялось. Не нужно учить всё за один день!";
  $("chat").innerHTML += `
`;
  const bubbles = $("chat").querySelectorAll(".bubble");
  bubbles[bubbles.length - 2].textContent = question;
  bubbles[bubbles.length - 1].textContent = answer;
  $("question").value = "";
}
$("chatForm").addEventListener("submit", e => {
  e.preventDefault();
  const q = $("question").value.trim();
  if (q) ask(q);
